1class Solution {
2    public int resilientSubarray(int[] nums, int k) {
3        int n = nums.length;
4        long[] prefix = new long[n + 1];
5        int[] remainder = new int[n];
6        for (int i = 0; i < n; i++) {
7            prefix[i + 1] = prefix[i] + nums[i];
8            remainder[i] = ((nums[i] % k) + k) % k;
9        }
10        int ans = 1;
11        int i = 0, j = 0;
12        while (j < n) {
13            if (remainder[i] == remainder[j]) {
14                long sum = prefix[j + 1] - prefix[i];
15                long mod = ((sum % k) + k) % k;
16                if (mod == remainder[i]) {
17                    ans = Math.max(ans, j - i + 1);
18                }
19                j++;
20            } else {
21                i = j;
22            }
23        }
24        return ans;
25    }
